114 template <typename Event>
115 void publish(const Event& event)
116 {
117 auto it = subscribers_.find(std::type_index(typeid(Event)));
118 if (it == subscribers_.end()) {
119 return;
120 }
121 const auto handlers = it->second; //> 快照，允许回调内退订
122 for (const auto& entry : handlers) {
123 entry.second(&event);
124 }
125 }

127private:
128 void unsubscribe(std::type_index type, SubscriptionId id)
129 {
130 auto it = subscribers_.find(type);
131 if (it == subscribers_.end()) {
132 return;
133 }
134 auto& handlers = it->second;
135 handlers.erase(
136 std::remove_if(handlers.begin(), handlers.end(),
137 [id](const auto& entry) { return entry.first == id; }),
138 handlers.end());
139 if (handlers.empty()) {
140 subscribers_.erase(it);
141 }
142 }

144 using ErasedHandler = std::function<void(const void*)>;
145 struct LifetimeToken { }; //> 仅作存活标记
146 std::unordered_map<std::type_index, std::vector<std::pair<SubscriptionId, ErasedHandler>>> subscribers_;
147 SubscriptionId next_id_ { 1 };
148 //> 生命周期令牌：订阅句柄持其弱引用探测总线存活。声明在最后（最先析构），
149 //> 使总线的析构过程尽早对所有句柄表现为已失效
