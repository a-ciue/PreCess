104 template <typename Event>
105 void publish(const Event& event)
106 {
107 auto it = subscribers_.find(std::type_index(typeid(Event)));
108 if (it == subscribers_.end()) {
109 return;
110 }
111 const auto handlers = it->second; //> 快照，允许回调内退订

118 void unsubscribe(std::type_index type, SubscriptionId id)
119 {
120 auto it = subscribers_.find(type);
121 if (it == subscribers_.end()) {
122 return;
123 }
124 auto& handlers = it->second;
125 handlers.erase(
126 std::remove_if(handlers.begin(), handlers.end(),
127 [id](const auto& entry) { return entry.first == id; }),
128 handlers.end());
129 if (handlers.empty()) {
130 subscribers_.erase(it);
131 }
132 }
