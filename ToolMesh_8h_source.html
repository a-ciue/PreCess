252{

265 CVertex* v = *fviter;

267 }
268 }
269 }

271
272 // file writer

274 std::ofstream out(file_name);
275
276 int vid = 1;
277 for (MeshVertexIterator viter(this); !viter.end(); viter++)
278 {
279 if ((*viter)->father() == 0) continue;
280 CVertex* v = *viter;
281 father_to_id[v->father()] = vid;
282 out << "Vertex " << vid << " " << v->point()[0] << " " << v->point()[1] << " " << v->point()[2] <<"{ father = ( "<<v->father()<<" )}" << std::endl;
283 vid++;
284 }

286 for (MeshFaceIterator fiter(this); !fiter.end(); fiter++)
287 {
288 if ((*fiter)->get_g() != id) continue;
289 CFace* f = *fiter;
