249inline void CToolMesh<V, E, F, H>::Patch_Write(int id, const std::string &path_prefix)
250{
251 for (MeshVertexIterator viter(this); !viter.end(); viter++)
252 {
253 CVertex* v = *viter;
254 v->father() = 0;
255 }
256 for (MeshFaceIterator fiter(this); !fiter.end(); fiter++)
257 {
258 CFace* f = *fiter;
259 if(f->get_g() == id)
260 {
261 for (FaceVertexIterator fviter(f); !fviter.end(); fviter++)
262 {
263 CVertex* v = *fviter;
264 v->father() = v->id();
265 }
266 }
267 }
268 std::unordered_map<int, int> father_to_id;
269
270 // file writer
271 std::string file_name = path_prefix + "_" + std::to_string(id) + ".m";
272 std::ofstream out(file_name);
273
274 int vid = 1;
275 for (MeshVertexIterator viter(this); !viter.end(); viter++)
276 {
277 if ((*viter)->father() == 0) continue;
278 CVertex* v = *viter;
279 father_to_id[v->father()] = vid;
280 out << "Vertex " << vid << " " << v->point()[0] << " " << v->point()[1] << " " << v->point()[2] <<"{ father = ( "<<v->father()<<" )}" << std::endl;
281 vid++;
282 }
283 vid = 1;
284 for (MeshFaceIterator fiter(this); !fiter.end(); fiter++)
285 {
286 if ((*fiter)->get_g() != id) continue;
287 CFace* f = *fiter;
288 if (f->get_g() == id)
289 {
290 out << "Face " << vid++;
291 for (FaceVertexIterator fviter(f); !fviter.end(); fviter++)
292 {
293 CVertex* v = *fviter;
294 out << " " << father_to_id[v->father()];
295 }
296 out << " {g=(" << f->get_g() << ")}"<<std::endl;
297 }
298 }
299 out.close();
300}
