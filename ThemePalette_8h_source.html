53inline void setThemeColor(QPalette& palette, QPalette::ColorRole role,
54 const QVariantMap& colors, const char* key)
55{
56 const QVariant value = colors.value(QLatin1String(key));
57 if (!value.isValid())
58 return;
59
60 const QColor color = value.canConvert<QColor>() ? value.value<QColor>()
61 : QColor(value.toString());
62 if (color.isValid())
63 palette.setColor(role, color);
64}

67inline QStringList missingThemeRoles(const QVariantMap& colors)
68{
69 QStringList missing;

78inline QPalette buildThemePalette(const QVariantMap& colors)
79{
80 QPalette palette;
