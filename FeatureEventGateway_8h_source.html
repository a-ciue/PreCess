44 template <typename Event>
45 [[nodiscard]] core::EventBus::Subscription subscribe(std::function<void(const Event&)> handler)
46 {
47 auto invoke = [handler = std::move(handler), model = model_, undo_stack = undo_stack_, owner = owner_](const Event& event) {
48 model->assertOwnerThread();
49 if constexpr (std::is_same_v<Event, ParameterChangedEvent>) {
50 if (event.feature != owner)
51 return;
52 }
53 invokePreview(handler, event, *model, undo_stack, owner);
54 };
55 if constexpr (std::is_same_v<Event, ModelEvent>) {
56 // 只有延后通知需要自持事件值与原订阅存活标记。
57 auto lifetime = std::make_shared<int>(0);
58 std::weak_ptr<int> alive = lifetime;
59 return bus_->subscribe<Event>(
60 [invoke = std::move(invoke), model = model_, defer = defer_,
61 lifetime = std::move(lifetime), alive](const Event& event) {
62 model->assertOwnerThread();
63 if (model->writesPending() && defer) {
64 defer([alive, invoke, event] {
65 if (!alive.expired())
66 invoke(event);
67 });
68 } else {
69 invoke(event);
70 }
71 });
72 } else {
73 return bus_->subscribe<Event>(std::move(invoke));
74 }
75 }
