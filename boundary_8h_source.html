148template<typename CVertex, typename CEdge, typename CFace, typename CHalfEdge>
149CLoop<CVertex, CEdge, CFace, CHalfEdge>::CLoop( CBaseMesh<CVertex, CEdge, CFace, CHalfEdge> * pMesh, CHalfEdge * pH )
150{
151 m_pMesh = pMesh;
152 m_pHalfedge = pH;
153
154 m_length = 0;
155 CHalfEdge * he = pH;
156 //trace the boundary loop
157 do{
158 CVertex * v = (CVertex*)he->target();
159 he = m_pMesh->vertexMostClwOutHalfEdge( v );
160 m_halfedges.push_back( he );
161 m_length += m_pMesh->edgeLength( (CEdge*)he->edge() );
162 }while( he != m_pHalfedge );
163}

180template<typename CVertex, typename CEdge, typename CFace, typename CHalfEdge>
181void CBoundary<CVertex, CEdge, CFace, CHalfEdge>::_bubble_sort( std::vector<CLoop<CVertex, CEdge, CFace, CHalfEdge>*> & loops)
182{
183 int i, j, flag = 1; // set flag to 1 to start first pass
184 CLoop<CVertex,CEdge,CFace,CHalfEdge> * temp; // holding variable
185 int numLength = (int)loops.size( );
186 for(i = 1; (i <= numLength) && flag; i++)
187 {
188 flag = 0;
189 for (j=0; j < (numLength -1); j++)
190 {
191 if (loops[j+1]->length() > loops[j]->length() ) // ascending order simply changes to <
192 {
193 temp = loops[j]; // swap elements
194 loops[j] = loops[j+1];
195 loops[j+1] = temp;
196 flag = 1; // indicates that a swap occurred.
197 }
198 }
199 }
200}

206template<typename CVertex, typename CEdge, typename CFace, typename CHalfEdge>
207CBoundary<CVertex, CEdge, CFace, CHalfEdge>::CBoundary( CBaseMesh<CVertex, CEdge, CFace, CHalfEdge> * pMesh )
208{
209 m_pMesh = pMesh;
210 //collect all boundary halfedges
211 std::set<CHalfEdge*> boundary_hes;
212 for( MeshEdgeIterator<CVertex, CEdge, CFace, CHalfEdge> eiter( m_pMesh); !eiter.end(); eiter ++ )
213 {
214 CEdge * e = *eiter;
215 if( !m_pMesh->isBoundary(e) ) continue;
216
217 CHalfEdge * he = m_pMesh->edgeHalfedge( e, 0);
218 boundary_hes.insert( he );
219 }
220 //trace all the boundary loops
221 while( !boundary_hes.empty() )
222 {
223 //get the first boundary halfedge
224 typename std::set<CHalfEdge*>::iterator siter = boundary_hes.begin();
225 CHalfEdge * he = *siter;
226 //trace along this boundary halfedge
227 CLoop<CVertex, CEdge, CFace, CHalfEdge> * pL = new CLoop<CVertex, CEdge, CFace, CHalfEdge>( m_pMesh, he );
228 assert(pL);
229 m_loops.push_back( pL );
230 //remove all the boundary halfedges, which are in the same boundary loop as the head, from the halfedge list
231 for( typename std::list<CHalfEdge*>::iterator hiter = pL->halfedges().begin();
232 hiter != pL->halfedges().end(); hiter ++ )
233 {
234 CHalfEdge * he = *hiter;
235 siter = boundary_hes.find( he );
236 if( siter == boundary_hes.end() ) continue;
237 boundary_hes.erase( siter );
238 }
239 }
240
241 //std::sort( vlps.begin(), vlps.end(), loop_compare<CVertex,CFace,CEdge,CHalfEdge> );
242 _bubble_sort( m_loops );
243}
