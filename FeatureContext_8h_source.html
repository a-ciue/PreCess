124 std::string label, Index component_id, Capture capture, Compute compute, Write write, bool masked = false)

126 using Input = std::decay_t<std::invoke_result_t<Capture, const ComponentOperator&>>;
127 using Result = std::decay_t<std::invoke_result_t<Compute, Input&, systems::job::ProgressFn>>;
128 // 访问阶段由 Runner 定序；一个载体同时持有输入与结果，不再分别共享。
129 struct Payload {
130 std::optional<Input> input;
131 std::optional<Result> result;
132 };
133 auto payload = std::make_shared<Payload>();
134 return runCapturedWriteback(std::move(label), component_id, [capture = std::move(capture), compute = std::move(compute), payload](const ComponentOperator& op) {
135 payload->input.emplace(capture(op));
136 return systems::job::JobTaskFn {
137 [compute, payload](systems::job::ProgressFn report) mutable {
138 payload->result.emplace(compute(*payload->input, std::move(report)));
139 }
140 }; }, [write = std::move(write), payload](ComponentOperator& op, systems::job::ProgressFn report) mutable { write(op, *payload->result, std::move(report)); }, masked);
