162inline CHalfEdge * CVertex::most_ccw_out_halfedge()
163{
164 //for interior vertex
165 if( !m_boundary )
166 {
167 return most_ccw_in_halfedge()->he_sym(); //most ccw out halfedge is the dual of the most ccw in halfedge
168 }
169
170 //for boundary vertex
171 CHalfEdge * he = m_halfedge->he_next();
172 //get the out halfedge which is the next halfedge of the most ccw in halfedge
173 CHalfEdge * ne = he->ccw_rotate_about_source();
174 //rotate ccwly around the source vertex
175 while( ne != NULL )
176 {
177 he = ne;
178 ne = he->ccw_rotate_about_source();
179 }
180
181 return he;
